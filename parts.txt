### 🏛️ Trennung von Spiel-Logik und DOM (TypeScript Architektur)

1. **Reine Spiel-Logik (`GameManager.ts` & Klassen):**
* Hält den aktuellen Spielzustand (`Player`, aktueller `Room`, aktiver `Puzzle`-Index, Punkte, Restzeit).
* Validiert Antworten und berechnet Punkte (z. B. $+100$ Punkte pro Rätsel, $-15$ pro Hinweis).
* Feuert Events oder Callback-Benachrichtigungen, wenn sich der Zustand ändert.


2. **DOM-Manipulation (`UIRenderer.ts` & `UIController.ts`):**
* Reagiert ausschließlich auf Events/Zustandsänderungen.
* Rendert die HTML-Elemente dynamisch (Formularvalidierung mit Bootstrap Feedback-Klassen `is-invalid` / `is-valid`, Reihenfolge-Buttons für Sequenz-Rätsel).
* Fängt Benutzer-Eingaben ab und übergibt sie an den `GameManager`.

### 🧠 5. TypeScript Architektur (Aufteilung nach Logik & DOM)

#### A. Modelle (`src/models/Player.ts`)

```typescript
export class Player {
  constructor(public name: string, public points: number = 500) {}

  addPoints(amount: number) {
    this.points += amount;
  }

  removePoints(amount: number) {
    this.points = Math.max(0, this.points - amount);
  }
}

```

```typescript
          selectedSequence.push(color);
          seqDisplay.textContent = `Auswahl: ${selectedSequence.join(' ➔ ')}`;
        });
      });
    }

    document.getElementById('puzzle-form')!.addEventListener('submit', (e) => {
      e.preventDefault();
      if (puzzle.type === 'sequence') {
        onAnswer(selectedSequence);
      } else {
        const val = (document.getElementById('puzzle-input') as HTMLInputElement).value;
        onAnswer(val);
      }
    });

    document.getElementById('hint-btn')!.addEventListener('click', () => {
      const hintDiv = document.getElementById('hint-container')!;
      hintDiv.innerHTML = `<div class="alert alert-warning py-2">💡 <strong>Hinweis:</strong> ${puzzle.hints[0]}</div>`;
      onHint();
    });

    document.getElementById('buy-time-btn')!.addEventListener('click', () => {
      onBuyTime();
    });
  }

  updateTimer(timeLeft: number) {
    const timerEl = document.getElementById('timer-val');
    if (timerEl) timerEl.textContent = timeLeft.toString();
  }

  renderVictory(points: number, onRestart: () => void) {
```

#### C. Haupt-Controller / Einstieg (`src/index.ts`)

```typescript
import { Player } from './models/Player';
import { UIRenderer } from './ui/UIRenderer';

class GameController {
  private renderer = new UIRenderer('app');
  private player!: Player;
  private roomsData: any = null;
  private currentRoom: any = null;
  private currentDifficulty: string = 'easy';
  private currentPuzzleIndex: number = 0;
  private timeLeft: number = 60;
  private timerInterval: any = null;

  async init() {
    // JSON-Daten laden
    const response = await fetch('src/data/rooms.json');
    this.roomsData = await response.json();

    this.renderer.renderStartScreen((name, roomId, difficulty) => {
      this.startGame(name, roomId, difficulty);
    });
  }

  private startGame(name: string, roomId: string, difficulty: string) {
    this.player = new Player(name, 500);
    this.currentDifficulty = difficulty;
    this.currentRoom = this.roomsData.rooms.find((r: any) => r.id === roomId);
    this.currentPuzzleIndex = 0;
    this.timeLeft = 90;

    this.startTimer();
    this.loadCurrentPuzzle();
  }

  private startTimer() {
    if (this.timerInterval) clearInterval(this.timerInterval);
    this.timerInterval = setInterval(() => {
      this.timeLeft--;
      this.renderer.updateTimer(this.timeLeft);

      if (this.timeLeft <= 0) {
        alert("⏰ Die Zeit ist abgelaufen! Du kannst dir aber mit Punkten Extra-Zeit kaufen, falls du welche hast.");
        this.timeLeft = 0;
      }
    }, 1000);
  }

  private loadCurrentPuzzle() {
    const puzzles = this.currentRoom.difficulties[this.currentDifficulty];
    if (this.currentPuzzleIndex >= puzzles.length) {
      clearInterval(this.timerInterval);
      this.renderer.renderVictory(this.player.points, () => this.init());
      return;
    }

    const puzzle = puzzles[this.currentPuzzleIndex];
    this.renderer.renderPuzzle(
      puzzle,
      this.player.points,
      this.timeLeft,
      (answer) => this.handleAnswer(answer, puzzle),
      () => {
        this.player.removePoints(50);
      },
      () => {
        if (this.player.points >= 100) {
          this.player.removePoints(100);
          this.timeLeft += 30;
          alert("⏱️ 30 Sekunden hinzugefügt!");
        } else {
          alert("❌ Zu wenig Punkte, um Zeit zu kaufen!");
        }
      }
    );
  }

  private handleAnswer(answer: any, puzzle: any) {
    let isCorrect = false;

    if (puzzle.type === 'sequence') {
      isCorrect = JSON.stringify(answer) === JSON.stringify(puzzle.solution);
    } else {
      isCorrect = answer.trim().toUpperCase() === puzzle.solution.toUpperCase();
    }

    if (isCorrect) {
      alert("✅ Richtig! Weiter zum nächsten Rätsel.");
      this.player.addPoints(100);
      this.currentPuzzleIndex++;
      this.loadCurrentPuzzle();
    } else {
      alert("❌ Leider falsch! Versuche es noch einmal.");
      this.player.removePoints(20);
    }
  }
}

// Spiel starten
new GameController().init();
```
